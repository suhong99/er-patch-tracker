## 데이터 구조 설계

```typescript
// 무기군별 스텟 (성장치가 무기군마다 다름)
type WeaponStat = {
  attackSpeedGrowth: number; // 공격속도 성장 (%)
  basicAttackAmpGrowth?: number; // 기본 공격 증폭 성장 (%)
  skillAmpGrowth?: number; // 스킬 증폭 성장 (%)
};

// 실험체 기본 스텟
type BaseStats = {
  // 체력
  hpBase: number;
  hpGrowth: number;
  // 방어력
  defenseBase: number;
  defenseGrowth: number;
  // 공격력
  attackBase: number;
  attackGrowth: number;
  // 이동속도
  moveSpeed: number;
  // 공격속도
  attackSpeedBase: number;
  // 무기군별 성장치
  weaponTypes: Record<string, WeaponStat>;
};
```
